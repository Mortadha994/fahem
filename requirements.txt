# Auth (Phase 0c). google-auth verifies Google ID tokens against Google's
# published keys. PyJWT signs/verifies our own session cookie: chosen over
# python-jose, which has had no release since 2021 and carried CVE-2024-33663
# / CVE-2024-33664 - PyJWT is narrower, actively maintained, and is what
# FastAPI's own security docs use.
# [requests] is not optional for us: google.auth.transport.requests, which
# verify_oauth2_token needs to fetch Google's public keys, raises ImportError
# at import time without it. Plain `google-auth` installs but cannot verify.
google-auth[requests]>=2.30.0
PyJWT>=2.8.0

# Rate limiting (Phase 2). slowapi is a thin FastAPI/Starlette adapter over
# `limits`, which does the window arithmetic and owns the Redis storage
# backend - so the counter logic is not hand-rolled here. Chosen over
# fastapi-limiter, whose last release was 2023: a stale dependency on the
# cost-control path is the same risk that ruled out python-jose in Phase 0c.
# redis is slowapi's storage driver, not a direct dependency of our code.
